# 公众号文章文档生成方法

本文档说明如何自动生成 XIAOFU工具箱公众号更新文章 Word 文档。

---

## 环境准备

### 1. 安装 Node.js
确保系统已安装 Node.js (推荐 v18+)

### 2. 安装 docx 库

```bash
npm install docx
```

---

## 生成脚本

创建 `create_doc.js` 文件：

```javascript
const { Document, Packer, Paragraph, TextRun, AlignmentType, HeadingLevel, LevelFormat } = require('docx');
const fs = require('fs');

const doc = new Document({
  // 文档样式配置

    default: { 
      document: { run: { font: "Microsoft YaHei", size: 24 } } // 12pt 默认字体
    },
    paragraphStyles: [
      // 标题样式
      { id: "Title", name: "Title", basedOn: "Normal",

        paragraph: { spacing: { before: 240, after: 120 }, alignment: AlignmentType.CENTER } },
      // 一级标题样式

        paragraph: { spacing: { before: 300, after: 150 }, outlineLevel: 0 } },
      // 二级标题样式

        run: { size: 28, bold: true, color: "388E3C", font: "Microsoft YaHei" },
        paragraph: { spacing: { before: 200, after: 100 }, outlineLevel: 1 } }
    ]
  },
  
  // 列表编号配置
  numbering: {
    config: [
      { reference: "bullet-list",
        levels: [{ level: 0, format: LevelFormat.BULLET, text: "•", alignment: AlignmentType.LEFT,
          style: { paragraph: { indent: { left: 720, hanging: 360 } } } }] },
      { reference: "v123-list",
        levels: [{ level: 0, format: LevelFormat.BULLET, text: "•", alignment: AlignmentType.LEFT,
          style: { paragraph: { indent: { left: 720, hanging: 360 } } } } ] },
      { reference: "v124-list",
        levels: [{ level: 0, format: LevelFormat.BULLET, text: "•", alignment: AlignmentType.LEFT,

      { reference: "kml-sub",
        levels: [{ level: 0, format: LevelFormat.BULLET, text: "◦", alignment: AlignmentType.LEFT,
          style: { paragraph: { indent: { left: 1080, hanging: 360 } } } } ] }

  },
  
  // 文档内容
  sections: [{
    properties: { 
      page: { margin: { top: 1440, right: 1440, bottom: 1440, left: 1440 } } // 1英寸边距
    },

      // 副标题

        alignment: AlignmentType.CENTER,

      }),

        spacing: { before: 200, after: 400 },
        children: [new TextRun({ text: "【整体工具截图位置】", color: "999999", italics: true, size: 28 })] 
      }),
      
      // 简介
      new Paragraph({ 
        alignment: AlignmentType.CENTER,
        spacing: { after: 400 },
        children: [new TextRun({ 
          text: "XIAOFU工具箱是基于 ArcGIS Pro SDK 开发的 GIS 扩展工具集，提供数据处理、转换、制图等功能", 
          size: 22, color: "555555"

      // v1.2.4 版本
      new Paragraph({ 
        heading: HeadingLevel.HEADING_1, 
        children: [new TextRun("📦 v1.2.4 版本更新")] 
      }),
      
      new Paragraph({ 
        heading: HeadingLevel.HEADING_2, 
        children: [new TextRun("🆕 新增功能")] 
      }),
      
      // 带编号的列表项
      new Paragraph({ 
        numbering: { reference: "v124-list", level: 0 },
        children: [new TextRun({ text: "要素图层分组导出KML/KMZ工具", bold: true })] 
      }),
      
      // 子项（缩进）
      new Paragraph({ 
        numbering: { reference: "kml-sub", level: 0 },
        children: [new TextRun("支持按字段分组导出或直接导出全部要素")] 
      }),
      
      // 截图占位
      new Paragraph({ 
        spacing: { before: 200 },
        children: [new TextRun({ text: "【工具界面截图位置】", color: "999999", italics: true })] 
      }),
      
      // 性能改进
      new Paragraph({ 
        heading: HeadingLevel.HEADING_2,

        alignment: AlignmentType.CENTER,
        spacing: { before: 200, after: 200 },

        })]

    ]

// 生成文档
Packer.toBuffer(doc).then(buffer => {
  fs.writeFileSync("Docs/公众号文档/XIAOFU工具箱_v1.2.3&v1.2.4_更新发布.docx", buffer);
  console.log("文档已生成");
});
```

## 运行生成

```bash
node create_doc.js
```

生成后的文档位于：`Docs/公众号文档/XIAOFU工具箱_v1.2.3&v1.2.4_更新发布.docx`

## 关键配置说明

### 字体设置
- 主字体：`Microsoft YaHei`（微软雅黑）
- 字号换算：size 值 = 实际字号(pt) × 2

### 常用字号对照
| size 值 | 实际字号 |
|---------|----------|
| 48 | 24pt（标题） |
| 32 | 16pt（一级标题） |
| 28 | 14pt（二级标题） |
| 24 | 12pt（正文） |
| 22 | 11pt（小字） |

### 间距设置
- 数值单位：twips（1/20 点）
- 常用值：1440 = 1英寸，200 = 约0.14英寸

### 颜色值
- 黑色：`000000`
- 绿色（主标题）：`2E7D32`
- 浅绿色（副标题）：`388E3C`
- 红色（强调）：`D32F2F`
- 灰色：`666666`、`999999`
